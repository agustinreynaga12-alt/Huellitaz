        // Componente de QR Dinámico en SVG
        const DynamicQRCodeSVG = ({ text, size = 200, color = "#047857", fgBg = "#ffffff" }) => {
          const gridSize = 21;
          
          const generateMatrix = (str) => {
            const matrix = Array(gridSize).fill(null).map(() => Array(gridSize).fill(false));
            
            const drawFinder = (startX, startY) => {
              for (let r = 0; r < 7; r++) {
                for (let c = 0; c < 7; c++) {
                  if (
                    r === 0 || r === 6 || c === 0 || c === 6 ||
                    (r >= 2 && r <= 4 && c >= 2 && c <= 4)
                  ) {
                    matrix[startY + r][startX + c] = true;
                  }
                }
              }
            };

            drawFinder(0, 0);
            drawFinder(gridSize - 7, 0);
            drawFinder(0, gridSize - 7);

            for (let i = 8; i < gridSize - 8; i++) {
              if (i % 2 === 0) {
                matrix[6][i] = true;
                matrix[i][6] = true;
              }
            }

            let hash = 0;
            for (let i = 0; i < str.length; i++) {
              hash = (hash << 5) - hash + str.charCodeAt(i);
              hash |= 0;
            }

            for (let r = 0; r < gridSize; r++) {
              for (let c = 0; c < gridSize; c++) {
                if (
                  (r < 8 && c < 8) ||
                  (r < 8 && c >= gridSize - 8) ||
                  (r >= gridSize - 8 && c < 8)
                ) {
                  continue;
                }

                const bit = Math.abs((hash ^ (r * 31 + c * 17) ^ (str.length * 7)) % 100);
                if (bit > 42) {
                  matrix[r][c] = true;
                }
              }
            }

            return matrix;
          };

          const matrix = useMemo(() => generateMatrix(text || "PET"), [text]);
          const cellSize = size / gridSize;

          return (
            <svg width={size} height={size} viewBox={`0 0 ${size} ${size}`} className="rounded-lg shadow-sm">
              <rect width={size} height={size} fill={fgBg} rx="8" />
              {matrix.map((row, rIdx) =>
                row.map((cell, cIdx) => {
                  if (!cell) return null;
                  return (
                    <rect
                      key={`${rIdx}-${cIdx}`}
                      x={cIdx * cellSize}
                      y={rIdx * cellSize}
                      width={cellSize + 0.3}
                      height={cellSize + 0.3}
                      fill={color}
                    />
                  );
                })
              )}
            </svg>
          );
        };

        const INITIAL_PETS = [
          {
            id: 'pet-1',
            name: 'Max',
            species: 'Perro',
            breed: 'Golden Retriever',
            birthDate: '2021-05-15',

            avatar: 'https://images.unsplash.com/photo-1552053831-71594a27632d?auto=format&fit=crop&q=80&w=300',
            ownerName: 'Carlos Mendoza',
            ownerPhone: '+54 9 11 4567 8901',
            ownerAddress: 'Av. Corrientes 1234, CABA',
            medicalAlerts: 'Alérgico a la Penicilina. Requiere medicación diaria.'
          }
        ];

          {
            id: 'rec-1',
            petId: 'pet-1',
            type: 'vaccine',
            title: 'Vacuna Antirrábica',
            productName: 'Rabisin Dosis 1ml',
            date: '2023-11-10',
            nextDueDate: '2026-12-10',

            notes: 'Aplicación en muslo derecho.'

        function App() {
          const [pets, setPets] = useState(INITIAL_PETS);
          const [records, setRecords] = useState(INITIAL_RECORDS);
          const [selectedPetId, setSelectedPetId] = useState('all');
          const [activeTab, setActiveTab] = useState('pets');
          
          const [selectedType, setSelectedType] = useState('all');
          const [searchQuery, setSearchQuery] = useState('');

          const [isPetModalOpen, setIsPetModalOpen] = useState(false);
          const [isRecordModalOpen, setIsRecordModalOpen] = useState(false);
          const [isScanPreviewOpen, setIsScanPreviewOpen] = useState(false);

          const [editingPet, setEditingPet] = useState(null);
          const [editingRecord, setEditingRecord] = useState(null);
          const [qrColor, setQrColor] = useState('#047857');

          const [petForm, setPetForm] = useState({
            name: '', species: 'Perro', breed: '', birthDate: '', gender: 'Macho',
            weight: '', chipNumber: '', avatar: '', ownerName: '', ownerPhone: '',
            ownerAddress: '', medicalAlerts: ''
          });

          const [recordForm, setRecordForm] = useState({
            petId: '', type: 'vaccine', title: '', productName: '',
            date: new Date().toISOString().split('T')[0], nextDueDate: '',
            veterinarian: '', cost: '', notes: ''
          });

          const calculateAge = (birthDateString) => {
            if (!birthDateString) return 'Desconocida';
            const birth = new Date(birthDateString);
            const now = new Date();
            let years = now.getFullYear() - birth.getFullYear();
            let months = now.getMonth() - birth.getMonth();
            if (months < 0 || (months === 0 && now.getDate() < birth.getDate())) {
              years--;
              months += 12;
            }
            if (years === 0) return `${months} ${months === 1 ? 'mes' : 'meses'}`;
            return `${years} ${years === 1 ? 'año' : 'años'}`;
          };

          const activePet = useMemo(() => {
            if (selectedPetId === 'all') return pets[0] || null;
            return pets.find(p => p.id === selectedPetId) || pets[0] || null;
          }, [pets, selectedPetId]);

          const handleOpenPetModal = (pet = null) => {
            if (pet) {
              setEditingPet(pet);
              setPetForm(pet);
            } else {
              setEditingPet(null);
              setPetForm({
                name: '', species: 'Perro', breed: '', birthDate: '', gender: 'Macho',
                weight: '', chipNumber: '', avatar: '', ownerName: '', ownerPhone: '',
                ownerAddress: '', medicalAlerts: ''
              });
            }
            setIsPetModalOpen(true);
          };

          const handleSavePet = (e) => {
            e.preventDefault();
            const avatarUrl = petForm.avatar.trim() || 'https://images.unsplash.com/photo-1543466835-00a7907e9de1?auto=format&fit=crop&q=80&w=300';
            if (editingPet) {
              setPets(pets.map(p => p.id === editingPet.id ? { ...petForm, id: editingPet.id, avatar: avatarUrl } : p));

              const newPet = { ...petForm, id: `pet-${Date.now()}`, avatar: avatarUrl };
              setPets([...pets, newPet]);
              setSelectedPetId(newPet.id);
            }
            setIsPetModalOpen(false);
          };

          const handleDeletePet = (petId) => {
            if (pets.length <= 1) {
              alert("Debes mantener al menos una mascota en el registro.");
              return;
            }
            setPets(pets.filter(p => p.id !== petId));
            setRecords(records.filter(r => r.petId !== petId));
            setSelectedPetId('all');
          };

          const handleOpenRecordModal = (record = null) => {
            if (record) {
              setEditingRecord(record);
              setRecordForm(record);
            } else {
              setEditingRecord(null);
              setRecordForm({
                petId: selectedPetId !== 'all' ? selectedPetId : (pets[0]?.id || ''),
                type: 'vaccine', title: '', productName: '',
                date: new Date().toISOString().split('T')[0], nextDueDate: '',
                veterinarian: '', cost: '', notes: ''
              });
            }
            setIsRecordModalOpen(true);
          };

          const handleSaveRecord = (e) => {
            e.preventDefault();
            if (editingRecord) {
              setRecords(records.map(r => r.id === editingRecord.id ? { ...recordForm, id: editingRecord.id } : r));
            } else {
              const newRecord = { ...recordForm, id: `rec-${Date.now()}` };
              setRecords([newRecord, ...records]);
            }
            setIsRecordModalOpen(false);
          };

          const handleDeleteRecord = (recordId) => {
            setRecords(records.filter(r => r.id !== recordId));
          };

          const filteredRecords = useMemo(() => {
            return records.filter(r => {
              const matchesPet = selectedPetId === 'all' || r.petId === selectedPetId;
              const matchesType = selectedType === 'all' || r.type === selectedType;
              const matchesSearch = 
                r.title.toLowerCase().includes(searchQuery.toLowerCase()) ||
                r.productName?.toLowerCase().includes(searchQuery.toLowerCase()) ||
                r.veterinarian?.toLowerCase().includes(searchQuery.toLowerCase());
              return matchesPet && matchesType && matchesSearch;
            });
          }, [records, selectedPetId, selectedType, searchQuery]);

            return records.filter(r => {
              if (!r.nextDueDate) return false;

              return diffDays <= 30;
            });

          const getTypeBadge = (type) => {

              case 'vaccine': return { label: 'Vacuna', bg: 'bg-emerald-100 text-emerald-800 border-emerald-200', icon: Syringe, iconBg: 'bg-emerald-600' };
              case 'deworming': return { label: 'Desparasitación', bg: 'bg-amber-100 text-amber-800 border-amber-200', icon: Bug, iconBg: 'bg-amber-600' };

                    </div>

                  <div className="flex space-x-2">
                    <button onClick={() => handleOpenPetModal()} className="bg-emerald-800 hover:bg-emerald-900 text-xs font-semibold px-3 py-2 rounded-lg flex items-center">

                      <Plus className="h-4 w-4 mr-1" /> Evento

                    </div>

                      </div>
                    </form>
                  </div>
                </div>
              )}

              {/* Modal Simulador */}
              {isScanPreviewOpen && activePet && (
                <div className="fixed inset-0 bg-slate-900/70 flex items-center justify-center p-4 z-50">
                  <div className="bg-white rounded-3xl p-6 max-w-xs w-full text-center space-y-4">

                  </div>
                </div>
              )}
            </div>
          );
        }
